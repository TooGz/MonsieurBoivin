{{- /* Vidéo locale accessible. Usage : {{< video src="medias/teaser.mp4" title="Description de la vidéo" >}} {{< video src="medias/delignage.mp4" title="…" animation="true" >}} ← remplace un GIF : lecture auto, muette, en boucle Paramètres : src (fichier du dossier de la page), title (description, obligatoire pour l'accessibilité), poster (image d'aperçu ; par défaut « -poster.jpg » à côté de la vidéo), animation, autoplay, loop, muted. - La source est résolue dans les ressources de la page (publication et chemin corrects). - Les dimensions de l'affiche réservent la place de la vidéo (pas de décalage de mise en page). - En mode animation, la lecture automatique est coupée si le visiteur a demandé à réduire les animations. */ -}} {{- $src := .Get "src" -}} {{- $title := .Get "title" | default "" -}} {{- $animation := eq (.Get "animation") "true" -}} {{- $autoplay := or $animation (eq (.Get "autoplay") "true") -}} {{- $loop := or $animation (eq (.Get "loop") "true") -}} {{- $muted := or $autoplay (eq (.Get "muted") "true") -}} {{- $clean := strings.TrimPrefix "./" $src -}} {{- $videoURL := $src -}} {{- with .Page.Resources.Get $clean }}{{ $videoURL = .RelPermalink }}{{ end -}} {{- $ext := path.Ext $clean | strings.TrimPrefix "." | lower -}} {{- $mime := index (dict "mp4" "video/mp4" "m4v" "video/mp4" "webm" "video/webm" "ogv" "video/ogg" "mov" "video/quicktime") $ext | default (printf "video/%s" $ext) -}} {{- $posterURL := "" -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- $posterName := .Get "poster" | default (printf "%s-poster.jpg" (strings.TrimSuffix (path.Ext $clean) $clean)) -}} {{- with .Page.Resources.Get (strings.TrimPrefix "./" $posterName) -}} {{- $img := . -}} {{- if gt .Width 1280 }}{{ $img = .Resize "1280x webp q78" }}{{ else }}{{ $img = .Process "webp q78" }}{{ end -}} {{- $posterURL = $img.RelPermalink -}} {{- $width = $img.Width -}} {{- $height = $img.Height -}} {{- else -}} {{- with .Get "poster" }}{{ $posterURL = . }}{{ end -}} {{- end -}} {{- /* Une seule fois par page (première vidéo) ; .Store persisterait entre deux rebuilds du serveur */ -}} {{- if eq .Ordinal 0 }} {{- end }}
